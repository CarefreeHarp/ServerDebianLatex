\subsubsection{Disponibilidad}
% Contenido pendiente de redacción.
La biblioteca de escenarios se guarda en el propio computador, de modo que la aplicación depende de internet solo para la consulta automática a Open-Meteo. Su disponibilidad se especifica con los siguientes requerimientos.

\begin{longtable}{|p{\rqA}|p{\rqB}|p{\rqC}|p{\rqD}|}
  \caption{RNF-06 -- Funcionar sin interrupciones}\label{tab:rnf-06}\tabularnewline
  \hline
  \textbf{\# Requerimiento} & RNF-06 & \raggedright\textbf{Tipo de requerimiento} & No funcional: disponibilidad\tabularnewline\hline
  \raggedright\textbf{Funcionalidad} & \multicolumn{3}{p{\rqBCD}|}{Todas las funcionalidades}\tabularnewline\hline
  \endfirsthead
  \textbf{Descripción} & \multicolumn{3}{p{\rqBCD}|}{La aplicación deberá estar disponible siempre que el computador esté encendido, sin periodos de mantenimiento ni un administrador que la atienda.}\tabularnewline\hline
  \textbf{Razón} & \multicolumn{3}{p{\rqBCD}|}{Una emergencia puede extenderse durante horas y requerir varias consultas sucesivas. Además, la aplicación se ejecuta localmente y no depende de un servicio en línea permanente.}\tabularnewline\hline
  \textbf{Autor} & \multicolumn{3}{p{\rqBCD}|}{el equipo de trabajo}\tabularnewline\hline
  \raggedright\textbf{Criterio de medición} & \multicolumn{3}{p{\rqBCD}|}{Prueba: se ejecutan 30~consultas de prueba consecutivas~\cite{vitoriano2021probabilidad} sin cerrar la aplicación, y todas se completan sin errores.}\tabularnewline\hline
  \textbf{Prioridad} & Alta & \raggedright\textbf{Módulo asociado} & Todos los módulos\tabularnewline\hline
  \textbf{Versión} & 1.0 & \textbf{Fecha} & 21/11/2026\tabularnewline\hline
\end{longtable}

\begin{longtable}{|p{\rqA}|p{\rqB}|p{\rqC}|p{\rqD}|}
  \caption{RNF-07 -- Operar sin conexión a internet}\label{tab:rnf-07}\tabularnewline
  \hline
  \textbf{\# Requerimiento} & RNF-07 & \raggedright\textbf{Tipo de requerimiento} & No funcional: disponibilidad\tabularnewline\hline
  \raggedright\textbf{Funcionalidad} & \multicolumn{3}{p{\rqBCD}|}{Obtener las condiciones meteorológicas}\tabularnewline\hline
  \endfirsthead
  \textbf{Descripción} & \multicolumn{3}{p{\rqBCD}|}{Todas las funciones de la aplicación deberán estar disponibles sin conexión a internet, excepto la consulta automática a Open-Meteo, que se reemplaza por el ingreso manual de las condiciones meteorológicas (RF-02).}\tabularnewline\hline
  \textbf{Razón} & \multicolumn{3}{p{\rqBCD}|}{El servicio gratuito de Open-Meteo no garantiza su disponibilidad~\cite{openmeteoterms} y durante una emergencia puede no haber conexión; por eso la biblioteca de escenarios se guarda en el propio computador.}\tabularnewline\hline
  \textbf{Autor} & \multicolumn{3}{p{\rqBCD}|}{el equipo de trabajo}\tabularnewline\hline
  \raggedright\textbf{Criterio de medición} & \multicolumn{3}{p{\rqBCD}|}{Prueba: con la conexión desactivada, el usuario completa una consulta con condiciones ingresadas manualmente y ve el resultado.}\tabularnewline\hline
  \textbf{Prioridad} & Alta & \raggedright\textbf{Módulo asociado} & Interfaz y visualización y Meteorología\tabularnewline\hline
  \textbf{Versión} & 1.0 & \textbf{Fecha} & 21/11/2026\tabularnewline\hline
\end{longtable}
